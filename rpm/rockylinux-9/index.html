Install

sudo curl -fsSL https://antoniofigueiredo.github.io/plxsdk/rpm/rockylinux-9/plxsdk.repo -o /etc/yum.repos.d/plxsdk.repo

sudo dnf makecache
sudo dnf install plxsdk

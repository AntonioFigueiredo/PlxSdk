Install

curl -fsSL https://antoniofigueiredo.github.io/plxsdk/apt/repo-public-key.asc | sudo gpg --dearmor -o /usr/share/keyrings/plxsdk-archive-keyring.gpg

echo "deb [signed-by=/usr/share/keyrings/plxsdk-archive-keyring.gpg] https://antoniofigueiredo.github.io/plxsdk/apt unstable main" | sudo tee /etc/apt/sources.list.d/plxsdk.list > /dev/null

sudo apt update
sudo apt install plxsdk-dkms
